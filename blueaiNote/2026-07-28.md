#!/usr/bin/env python3
"""TikTok Ads 广告创建脚本（Business API v1.3 / REST）"""
import sys, json
import httpx

token = sys.argv[1]
advertiser_id = sys.argv[2]

BASE = "https://business-api.tiktok.com/open_api/v1.3"
HEADERS = {"Access-Token": token, "Content-Type": "application/json"}

result = {"steps": {}}


def post(path, body):
    resp = httpx.post(f"{BASE}{path}", headers=HEADERS, json=body, timeout=30)
    data = resp.json()
    return data


def get(path, params):
    resp = httpx.get(f"{BASE}{path}", headers=HEADERS, params=params, timeout=30)
    try:
        return resp.json()
    except Exception:
        return {"code": -1, "message": f"non-json response, status={resp.status_code}"}


# ── 0. 解析地区：South Korea ──────────────────────────────────────────────
loc_resp = get("/tool/region/", {"advertiser_id": advertiser_id, "placements": json.dumps(["PLACEMENT_TIKTOK"]), "objective_type": "APP_PROMOTION"})
result["steps"]["region_lookup"] = loc_resp if loc_resp.get("code") != 0 else {"code": 0, "message": "OK"}
location_id = None
if loc_resp.get("code") == 0:
    for region in loc_resp.get("data", {}).get("region_info", []):
        name = (region.get("name") or "").lower()
        if name in ("south korea", "korea, republic of", "korea", "republic of korea"):
            location_id = region.get("location_id") or region.get("region_code") or region.get("id")
            break
result["location_id"] = location_id

# ── 1. 解析推广应用 App ID ────────────────────────────────────────────────
app_resp = get("/app/list/", {"advertiser_id": advertiser_id})
result["steps"]["app_list"] = app_resp if app_resp.get("code") != 0 else {"code": 0, "message": "OK"}
tiktok_app_id = None
if app_resp.get("code") == 0:
    for app in app_resp.get("data", {}).get("list", []):
        pkg = app.get("package_name") or app.get("app_name") or ""
        if "catcafe2" in str(pkg).lower() or "com.gamesofa.android.catcafe2" == pkg:
            tiktok_app_id = app.get("app_id")
            break
result["tiktok_app_id"] = tiktok_app_id


# ── 2. 解析已授权发布身份 identity ────────────────────────────────────────
identity_resp = get("/identity/get/", {"advertiser_id": advertiser_id})
result["steps"]["identity_get"] = {"code": identity_resp.get("code"), "message": identity_resp.get("message")}
identity_type = None
identity_id = None
if identity_resp.get("code") == 0:
    identities = identity_resp.get("data", {}).get("identity_list", [])
    if identities:
        identity_type = identities[0].get("identity_type")
        identity_id = identities[0].get("identity_id")
result["identity_type"] = identity_type
result["identity_id"] = identity_id

# ── 3. 创意图片：直接复用第一次成功上传得到的 image_id（避免重复上传报重名错误）──
result["steps"]["image_upload"] = {"code": 0, "message": "reused existing upload"}
image_id = "ad-site-i18n-sg/20260728c7c7f4e62326a8ef455a8a71"
result["image_id"] = image_id

# ── 4. 兴趣类别检索：games（已人工确认 Games 一级类目 id=25，直接复用）─────────
interest_resp = get("/tool/interest_category/", {"advertiser_id": advertiser_id})
interest_category_id = None
if interest_resp.get("code") == 0:
    cats = interest_resp.get("data", {}).get("interest_categories") or interest_resp.get("data", {}).get("category_list") or interest_resp.get("data", {}).get("categories") or []
    for c in cats:
        cname = str(c.get("name") or c.get("category_name") or "").lower()
        if "game" in cname:
            interest_category_id = c.get("id") or c.get("category_id") or c.get("interest_category_id")
            break
result["steps"]["interest_category"] = {"code": interest_resp.get("code"), "message": interest_resp.get("message")}
result["interest_category_id"] = interest_category_id or "25"

print(json.dumps(result, ensure_ascii=False))
